@Composable
fun DiceWithButtonAndImage(modifier: Modifier = Modifier) {
    var result by remember { mutableStateOf(1) }
    val imageResource = when(result) {
        1 -> R.drawable.dice_1
        2 -> R.drawable.dice_2
        3 -> R.drawable.dice_3
        4 -> R.drawable.dice_4
        5 -> R.drawable.dice_5
        else -> R.drawable.dice_6
    }
    Column(modifier = modifier, horizontalAlignment = Alignment.CenterHorizontally) {
        Image(painter = painterResource(imageResource), contentDescription = result.toString())

        Spacer(modifier = Modifier.height(16.dp))

        Button(
            onClick = { result = (1..6).random() }
        ) {
            Text(text = stringResource(R.string.roll), fontSize = 24.sp)
        }
    }
}

Catatan perbaikan

Kode ini awalnya error karena resource-nya belum lengkap: gambar dadu (dice_1 sampai dice_6) belum ditambahkan ke res/drawable, dan string roll belum didefinisikan di strings.xml. Setelah kedua resource itu ditambahkan dengan nama yang persis sama, aplikasi berjalan normal.

Student Manager

Ringkasan

Aplikasi pengelolaan data mahasiswa (Student Manager) yang dibangun sebagai Mini Project. Aplikasi ini mencakup fitur Splash Screen otomatis, daftar mahasiswa dengan pencarian (Search), form penambahan dan pengubahan data dengan Dropdown Menu, serta dialog konfirmasi penghapusan. Tampilan UI dibangun menggunakan Scaffold, TopAppBar, dan FloatingActionButton agar persis seperti wireframe, sementara datanya murni dikelola menggunakan state lokal (remember { mutableStateOf() }).

Komponen yang digunakan

Scaffold & TopAppBarKerangka utama aplikasi yang menampung header, menu navigasi, dan tombol aksi mengambang.
LazyColumn & CardMenampilkan daftar data mahasiswa secara efisien dan rapi dalam bentuk kartu.
AlertDialogMenampilkan pop-up konfirmasi sebelum menghapus data mahasiswa dan info tentang aplikasi.
ExposedDropdownMenuBoxMembuat form pilihan (dropdown) interaktif untuk Program Studi.

        mutableStateOf(

    }

    var studentToEdit by remember { mutableStateOf<Student?>(null) }

    when (currentScreen) {
        "splash" -> SplashScreen(onFinish = { currentScreen = "home" })
        "home" -> HomeScreen(
            students = students,
            onAdd = { 
                studentToEdit = null
                currentScreen = "form" 
            },
            onEdit = { student ->
                studentToEdit = student
                currentScreen = "form"
            },
            onDelete = { student ->
                students = students.filterNot { it.id == student.id }
            },
            onRefresh = { /* Aksi kosong */ }
        )
        "form" -> FormScreen(
            initial = studentToEdit,
            existingStudents = students,
            onBack = { currentScreen = "home" },
            onSave = { nim, name, major ->
                if (studentToEdit == null) {
                    students = students + Student(nextId++, nim, name, major)
                } else {
                    students = students.map {
                        if (it.id == studentToEdit!!.id) it.copy(nim = nim, name = name, major = major) else it
                    }
                }
                currentScreen = "home"
            }
        )
    }

fun FormScreen(

                singleLine = true
            )
            ExposedDropdownMenuBox(expanded = expanded, onExpandedChange = { expanded = it }) {
                OutlinedTextField(
                    value = major,
                    onValueChange = {},
                    readOnly = true,
                    label = { Text("Program Studi") },
                    placeholder = { Text("Pilih program studi") },
                    trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expanded) },
                    modifier = Modifier.menuAnchor().fillMaxWidth()
                )
                ExposedDropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
                    majors.forEach { selection ->
                        DropdownMenuItem(
                            text = { Text(selection) },
                            onClick = { major = selection; expanded = false; error = null }
                        )
                    }
                }
            }
            error?.let { Text(it, color = MaterialTheme.colorScheme.error, fontSize = 13.sp) }
            
            Spacer(Modifier.weight(1f))
            
            Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {

                    onClick = {

                            nim.isBlank() -> error = "NIM wajib diisi"

                            name.isBlank() -> error = "Nama wajib diisi"
                            major.isBlank() -> error = "Program studi wajib dipilih"
                            existingStudents.any { it.nim == nim && it.id != initial?.id } -> error = "NIM sudah terdaftar"
                            else -> onSave(nim.trim(), name.trim(), major)
                        }
                    },
                    modifier = Modifier.weight(1f),
                    colors = ButtonDefaults.buttonColors(containerColor = Color.Blue)
                ) { Text("Simpan") }
            }
        }
    }

LibraryReg: Aplikasi Pendaftaran Anggota Perpustakaan

Deskripsi sistem

Sebuah perpustakaan perguruan tinggi masih melakukan pendaftaran anggota secara manual menggunakan formulir kertas. Proses tersebut menimbulkan beberapa kendala, seperti antrean pendaftaran, kesalahan pencatatan data, kesulitan mencari informasi anggota, dan lambatnya pembuatan kartu anggota.

Untuk mengatasi masalah tersebut, dibangun aplikasi Android bernama LibraryReg, aplikasi pendaftaran anggota perpustakaan yang memungkinkan calon anggota mengisi formulir pendaftaran melalui smartphone. Petugas juga dapat melihat daftar anggota yang telah terdaftar, mencari anggota berdasarkan nama atau nomor induk, serta mengubah dan menghapus data sesuai kebutuhan.

Ruang lingkup aplikasi

Formulir pendaftaranCalon anggota mengisi nama lengkap, NIM/NIP, program studi, alamat, nomor telepon, email, dan jenis keanggotaan (Mahasiswa, Dosen, Umum).
Validasi dataMengecek kolom wajib, format email, format nomor telepon, dan NIM/NRP yang sudah terdaftar sebelum data disimpan.
Manajemen anggotaMenampilkan daftar anggota, melihat detail, mencari berdasarkan nama atau NIM, mengedit, dan menghapus data dengan dialog konfirmasi.
Statistik ringkasMenampilkan jumlah total anggota serta rincian per jenis keanggotaan (Mahasiswa, Dosen, Umum) di bagian atas layar.
Penyimpanan lokalData anggota disimpan di perangkat lewat SharedPreferences dalam format JSON, sehingga tetap ada saat aplikasi ditutup dan dibuka kembali.

Di luar ruang lingkup: aplikasi ini belum terhubung ke server atau basis data pusat, tidak ada autentikasi petugas, dan belum mencetak kartu anggota fisik.

Komponen yang digunakan

Scaffold & FloatingActionButtonKerangka layar utama dengan tombol tambah anggota mengambang.
LazyColumn & CardMenampilkan daftar anggota secara efisien dalam bentuk kartu.
Dialog & AlertDialogForm pendaftaran/edit anggota, detail anggota, serta konfirmasi hapus data.
OutlinedTextField & DropdownMenuInput data anggota dan pemilihan jenis keanggotaan.
SharedPreferences + JSONMenyimpan dan memuat data anggota secara lokal lewat JSONArray/JSONObject.
State (remember / mutableStateListOf)Menyimpan daftar anggota, kata kunci pencarian, dan status dialog yang sedang terbuka.

private fun loadAnggota(context: Context): List<Anggota> {
    return try {
        val raw = context.getSharedPreferences(
            "libraryreg", Context.MODE_PRIVATE
        ).getString("anggota", "[]") ?: "[]"

        val json = JSONArray(raw)
        (0 until json.length()).map { i ->
            val item = json.getJSONObject(i)
            Anggota(
                nama = item.optString("nama"),
                nim = item.optString("nim"),
                prodi = item.optString("prodi"),
                alamat = item.optString("alamat"),
                telepon = item.optString("telepon"),
                email = item.optString("email"),
                jenis = item.optString("jenis", "Mahasiswa")
            )
        }
    } catch (_: Exception) {
        emptyList()
    }
}

private fun saveAnggota(context: Context, data: List<Anggota>) {
    val json = JSONArray()
    data.forEach { a ->
        json.put(
            JSONObject().apply {
                put("nama", a.nama)
                put("nim", a.nim)
                put("prodi", a.prodi)
                put("alamat", a.alamat)
                put("telepon", a.telepon)
                put("email", a.email)
                put("jenis", a.jenis)
            }
        )
    }

    context.getSharedPreferences("libraryreg", Context.MODE_PRIVATE)
        .edit()
        .putString("anggota", json.toString())
        .apply()
}

                                !android.util.Patterns.EMAIL_ADDRESS.matcher(email.trim()).matches() ->
                                    error = "Format email tidak valid."
                                !telepon.trim().matches(Regex("[0-9+() -]{8,20}")) ->
                                    error = "Nomor telepon tidak valid."
                                daftar.any {
                                    it.nim.equals(nim.trim(), ignoreCase = true) &&
                                            !it.nim.equals(
                                                anggotaAwal?.nim,
                                                ignoreCase = true
                                            )
                                } -> error = "NIM/NRP sudah digunakan anggota lain."
                                else -> onSimpan(
                                    Anggota(
                                        nama.trim(), nim.trim(), prodi.trim(),
                                        alamat.trim(), telepon.trim(), email.trim(), jenis
                                    )
                                )
                            }
                        },
                        modifier = Modifier.weight(1f),
                        shape = RoundedCornerShape(50)
                    ) {
                        Icon(Icons.Default.CheckCircle, null, modifier = Modifier.size(18.dp))
                        Spacer(Modifier.width(6.dp))
                        Text("Simpan")
                    }
                }
            }
        }
    }
}
