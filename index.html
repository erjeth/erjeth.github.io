private val GradientTop = Color(0xFFFF6B9D)
private val GradientBottom = Color(0xFFFFC371)

@Composable
fun BirthdayApp() {
    var name by remember { mutableStateOf("") }
    var showMessage by remember { mutableStateOf(false) }

    val infiniteTransition = rememberInfiniteTransition(label = "cake-bounce")
    val bounce by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = -14f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = 700, easing = FastOutSlowInEasing),
            repeatMode = RepeatMode.Reverse
        ),
        label = "bounce"
    )

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Brush.verticalGradient(listOf(GradientTop, GradientBottom)))
            .padding(24.dp),
        contentAlignment = Alignment.Center
    ) {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(text = "\uD83C\uDF88  \uD83C\uDF89  \uD83C\uDF88", fontSize = 32.sp)
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = "\uD83C\uDF82",
                fontSize = 72.sp,
                modifier = Modifier.offset(y = bounce.dp)
            )
            Spacer(modifier = Modifier.height(12.dp))
            Text(
                text = "Selamat Ulang Tahun!",
                fontSize = 28.sp,
                fontWeight = FontWeight.Bold,
                color = Color.White,
                textAlign = TextAlign.Center
            )
            Spacer(modifier = Modifier.height(24.dp))

            Card(
                shape = RoundedCornerShape(20.dp),
                colors = CardDefaults.cardColors(containerColor = Color.White.copy(alpha = 0.94f)),
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(modifier = Modifier.padding(20.dp)) {
                    Text(text = "Untuk siapa ucapan ini?", fontWeight = FontWeight.Medium, fontSize = 14.sp)
                    Spacer(modifier = Modifier.height(8.dp))
                    OutlinedTextField(
                        value = name,
                        onValueChange = { name = it; showMessage = false },
                        placeholder = { Text("Masukkan nama") },
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                        modifier = Modifier.fillMaxWidth()
                    )
                    Spacer(modifier = Modifier.height(16.dp))
                    Button(
                        onClick = { showMessage = true },
                        enabled = name.isNotBlank(),
                        modifier = Modifier.fillMaxWidth(),
                        colors = ButtonDefaults.buttonColors(containerColor = GradientTop)
                    ) {
                        Text("Kirim Ucapan \uD83C\uDF81")
                    }

                    AnimatedVisibility(visible = showMessage) {
                        Column(modifier = Modifier.padding(top = 16.dp)) {
                            Divider()
                            Spacer(modifier = Modifier.height(12.dp))
                            Text(text = "Selamat ulang tahun, $name! \uD83C\uDF89", fontWeight = FontWeight.Bold, fontSize = 18.sp)
                            Spacer(modifier = Modifier.height(4.dp))
                            Text(
                                text = "Semoga panjang umur, sehat selalu, dan semua harapan tahun ini terwujud. Selamat merayakan! \uD83E\uDD73",
                                fontSize = 14.sp,
                                color = Color.DarkGray
                            )
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun DiceWithButtonAndImage(modifier: Modifier = Modifier) {
    var result by remember { mutableStateOf(1) }
    val imageResource = when(result) {
        1 -> R.drawable.dice_1
        2 -> R.drawable.dice_2
        3 -> R.drawable.dice_3
        4 -> R.drawable.dice_4
        5 -> R.drawable.dice_5
        else -> R.drawable.dice_6
    }
    Column(modifier = modifier, horizontalAlignment = Alignment.CenterHorizontally) {
        Image(painter = painterResource(imageResource), contentDescription = result.toString())

        Spacer(modifier = Modifier.height(16.dp))

        Button(
            onClick = { result = (1..6).random() }
        ) {
            Text(text = stringResource(R.string.roll), fontSize = 24.sp)
        }
    }
}

Catatan perbaikan

Kode ini awalnya error karena resource-nya belum lengkap: gambar dadu (dice_1 sampai dice_6) belum ditambahkan ke res/drawable, dan string roll belum didefinisikan di strings.xml. Setelah kedua resource itu ditambahkan dengan nama yang persis sama, aplikasi berjalan normal.

Student Manager

Ringkasan

Aplikasi pengelolaan data mahasiswa (Student Manager) yang dibangun sebagai Mini Project. Aplikasi ini mencakup fitur Splash Screen otomatis, daftar mahasiswa dengan pencarian (Search), form penambahan dan pengubahan data dengan Dropdown Menu, serta dialog konfirmasi penghapusan. Tampilan UI dibangun menggunakan Scaffold, TopAppBar, dan FloatingActionButton agar persis seperti wireframe, sementara datanya murni dikelola menggunakan state lokal (remember { mutableStateOf() }).

Komponen yang digunakan

Scaffold & TopAppBarKerangka utama aplikasi yang menampung header, menu navigasi, dan tombol aksi mengambang.
LazyColumn & CardMenampilkan daftar data mahasiswa secara efisien dan rapi dalam bentuk kartu.
AlertDialogMenampilkan pop-up konfirmasi sebelum menghapus data mahasiswa dan info tentang aplikasi.
ExposedDropdownMenuBoxMembuat form pilihan (dropdown) interaktif untuk Program Studi.

@Composable
fun StudentApp() {
    var currentScreen by remember { mutableStateOf("splash") }
    var students by remember {
        mutableStateOf(
            listOf(
                Student(1, "2301001", "Budi Santoso", "Informatika"),
                Student(2, "2301002", "Siti Aminah", "Sistem Informasi"),
                Student(3, "2301003", "Andi Wijaya", "Teknik Komputer")
            )
        )
    }
    var nextId by remember { mutableStateOf(4) }
    var studentToEdit by remember { mutableStateOf<Student?>(null) }

    when (currentScreen) {
        "splash" -> SplashScreen(onFinish = { currentScreen = "home" })
        "home" -> HomeScreen(
            students = students,
            onAdd = { 
                studentToEdit = null
                currentScreen = "form" 
            },
            onEdit = { student ->
                studentToEdit = student
                currentScreen = "form"
            },
            onDelete = { student ->
                students = students.filterNot { it.id == student.id }
            },
            onRefresh = { /* Aksi kosong */ }
        )
        "form" -> FormScreen(
            initial = studentToEdit,
            existingStudents = students,
            onBack = { currentScreen = "home" },
            onSave = { nim, name, major ->
                if (studentToEdit == null) {
                    students = students + Student(nextId++, nim, name, major)
                } else {
                    students = students.map {
                        if (it.id == studentToEdit!!.id) it.copy(nim = nim, name = name, major = major) else it
                    }
                }
                currentScreen = "home"
            }
        )
    }
}

        Text("Kelola data mahasiswa dengan mudah", fontSize = 13.sp, color = Color.Gray)
        Spacer(Modifier.height(24.dp))
        CircularProgressIndicator()
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun HomeScreen(
    students: List<Student>,
    onAdd: () -> Unit,
    onEdit: (Student) -> Unit,
    onDelete: (Student) -> Unit,
    onRefresh: () -> Unit
) {
    var query by remember { mutableStateOf("") }
    var menuOpen by remember { mutableStateOf(false) }
    var showAbout by remember { mutableStateOf(false) }
    var studentToDelete by remember { mutableStateOf<Student?>(null) }
    val activity = LocalContext.current as? Activity

    val filteredStudents = if (query.isBlank()) students else students.filter {
        it.name.contains(query, true) || it.nim.contains(query, true) || it.major.contains(query, true)
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Student Manager") },
                actions = {
                    IconButton(onClick = { menuOpen = true }) { Icon(Icons.Default.MoreVert, "Menu") }
                    DropdownMenu(expanded = menuOpen, onDismissRequest = { menuOpen = false }) {
                        DropdownMenuItem(
                            text = { Text("Refresh") },
                            leadingIcon = { Icon(Icons.Default.Refresh, null) },
                            onClick = { query = ""; menuOpen = false }
                        )
                        DropdownMenuItem(
                            text = { Text("Tentang Aplikasi") },
                            leadingIcon = { Icon(Icons.Default.Info, null) },
                            onClick = { showAbout = true; menuOpen = false }
                        )
                        DropdownMenuItem(
                            text = { Text("Keluar") },
                            leadingIcon = { Icon(Icons.AutoMirrored.Filled.ExitToApp, null) },
                            onClick = { menuOpen = false; activity?.finish() }
                        )
                    }
                }
            )
        },
        floatingActionButton = {
            FloatingActionButton(onClick = onAdd, containerColor = Color.Blue, contentColor = Color.White) { 
                Icon(Icons.Default.Add, "Tambah") 
            }
        }
    ) { padding ->
        Column(Modifier.padding(padding).padding(horizontal = 16.dp)) {
            OutlinedTextField(
                value = query,
                onValueChange = { query = it },
                modifier = Modifier.fillMaxWidth(),
                placeholder = { Text("Cari mahasiswa...") },
                leadingIcon = { Icon(Icons.Default.Search, null) },
                trailingIcon = {
                    if (query.isNotEmpty()) {
                        IconButton(onClick = { query = "" }) { Icon(Icons.Default.Clear, "Hapus") }
                    }
                },
                singleLine = true
            )
            Spacer(Modifier.height(12.dp))
            Text("Jumlah mahasiswa: ${filteredStudents.size}", fontSize = 13.sp, fontWeight = FontWeight.Medium)
            Spacer(Modifier.height(8.dp))
            
            if (filteredStudents.isEmpty()) {
                Column(
                    modifier = Modifier.fillMaxSize(),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    verticalArrangement = Arrangement.Center
                ) {
                    Icon(Icons.Default.School, null, modifier = Modifier.size(72.dp), tint = Color.LightGray)
                    Spacer(Modifier.height(12.dp))
                    Text(
                        if (query.isNotEmpty()) "Data tidak ditemukan" else "Belum ada data mahasiswa",
                        fontWeight = FontWeight.Bold
                    )
                    if (query.isEmpty()) {
                        Text(
                            "Tekan tombol + untuk menambahkan mahasiswa pertama.",
                            fontSize = 13.sp, color = Color.Gray, textAlign = TextAlign.Center
                        )
                    }
                }
            } else {
                LazyColumn(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                    items(filteredStudents, key = { it.id }) { student ->
                        Card(modifier = Modifier.fillMaxWidth()) {
                            Row(Modifier.padding(12.dp), verticalAlignment = Alignment.CenterVertically) {
                                Icon(Icons.Default.Person, null, modifier = Modifier.size(40.dp), tint = Color.Gray)
                                Spacer(Modifier.width(12.dp))
                                Column(Modifier.weight(1f)) {
                                    Text(student.name, fontWeight = FontWeight.Bold)
                                    Text("NIM: ${student.nim}", fontSize = 12.sp, color = Color.Gray)
                                    Text(student.major, fontSize = 12.sp, color = Color.Gray)
                                }
                                IconButton(onClick = { onEdit(student) }) { 
                                    Icon(Icons.Default.Edit, "Edit", modifier = Modifier.size(20.dp), tint = Color.Blue) 
                                }
                                IconButton(onClick = { studentToDelete = student }) {
                                    Icon(Icons.Default.Delete, "Hapus", tint = Color.Red, modifier = Modifier.size(20.dp))
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    studentToDelete?.let { student ->
        AlertDialog(
            onDismissRequest = { studentToDelete = null },
            icon = { Icon(Icons.Default.Delete, null, tint = Color.Red) },
            title = { Text("Hapus Mahasiswa?") },
            text = { Text("Apakah Anda yakin ingin menghapus data ${student.name}?") },
            confirmButton = {
                Button(
                    onClick = { onDelete(student); studentToDelete = null },
                    colors = ButtonDefaults.buttonColors(containerColor = Color.Red)
                ) { Text("Hapus") }
            },
            dismissButton = { OutlinedButton(onClick = { studentToDelete = null }) { Text("Batal") } }
        )
    }

    if (showAbout) {
        AlertDialog(
            onDismissRequest = { showAbout = false },
            title = { Text("Tentang Aplikasi") },
            text = { Text("Student Manager v1.0\nAplikasi sederhana untuk mengelola data mahasiswa.") },
            confirmButton = { TextButton(onClick = { showAbout = false }) { Text("Tutup") } }
        )
    }

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun FormScreen(
    initial: Student?,
    existingStudents: List<Student>,
    onBack: () -> Unit,
    onSave: (String, String, String) -> Unit
) {
    var nim by remember { mutableStateOf(initial?.nim ?: "") }
    var name by remember { mutableStateOf(initial?.name ?: "") }
    var major by remember { mutableStateOf(initial?.major ?: "") }
    var expanded by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text(if (initial == null) "Tambah Mahasiswa" else "Edit Mahasiswa") },
                navigationIcon = {
                    IconButton(onClick = onBack) { Icon(Icons.AutoMirrored.Filled.ArrowBack, "Kembali") }
                }
            )
        }
    ) { padding ->
        Column(
            Modifier.padding(padding).padding(16.dp).fillMaxSize(),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            OutlinedTextField(
                value = nim,
                onValueChange = { nim = it; error = null },
                label = { Text("NIM") },
                placeholder = { Text("Masukkan NIM") },
                modifier = Modifier.fillMaxWidth(),
                singleLine = true
            )
            OutlinedTextField(
                value = name,
                onValueChange = { name = it; error = null },
                label = { Text("Nama") },
                placeholder = { Text("Masukkan nama mahasiswa") },
                modifier = Modifier.fillMaxWidth(),
                singleLine = true
            )
            ExposedDropdownMenuBox(expanded = expanded, onExpandedChange = { expanded = it }) {
                OutlinedTextField(
                    value = major,
                    onValueChange = {},
                    readOnly = true,
                    label = { Text("Program Studi") },
                    placeholder = { Text("Pilih program studi") },
                    trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expanded) },
                    modifier = Modifier.menuAnchor().fillMaxWidth()
                )
                ExposedDropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
                    majors.forEach { selection ->
                        DropdownMenuItem(
                            text = { Text(selection) },
                            onClick = { major = selection; expanded = false; error = null }
                        )
                    }
                }
            }
            error?.let { Text(it, color = MaterialTheme.colorScheme.error, fontSize = 13.sp) }
            
            Spacer(Modifier.weight(1f))
            
            Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                OutlinedButton(onClick = onBack, modifier = Modifier.weight(1f)) { Text("Batal") }
                Button(
                    onClick = {
                        when {
                            nim.isBlank() -> error = "NIM wajib diisi"
                            !nim.all { it.isDigit() } -> error = "NIM hanya boleh angka"
                            name.isBlank() -> error = "Nama wajib diisi"
                            major.isBlank() -> error = "Program studi wajib dipilih"
                            existingStudents.any { it.nim == nim && it.id != initial?.id } -> error = "NIM sudah terdaftar"
                            else -> onSave(nim.trim(), name.trim(), major)
                        }
                    },
                    modifier = Modifier.weight(1f),
                    colors = ButtonDefaults.buttonColors(containerColor = Color.Blue)
                ) { Text("Simpan") }
            }
        }
    }
}
