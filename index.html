@Composable
fun BirthdayApp() {
    var name by remember { mutableStateOf("") }
    var showMessage by remember { mutableStateOf(false) }

    val infiniteTransition = rememberInfiniteTransition(label = "cake-bounce")
    val bounce by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = -14f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = 700, easing = FastOutSlowInEasing),
            repeatMode = RepeatMode.Reverse
        ),
        label = "bounce"
    )

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Brush.verticalGradient(listOf(GradientTop, GradientBottom)))
            .padding(24.dp),
        contentAlignment = Alignment.Center
    ) {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(text = "\uD83C\uDF88  \uD83C\uDF89  \uD83C\uDF88", fontSize = 32.sp)
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = "\uD83C\uDF82",
                fontSize = 72.sp,
                modifier = Modifier.offset(y = bounce.dp)
            )
            Spacer(modifier = Modifier.height(12.dp))
            Text(
                text = "Selamat Ulang Tahun!",
                fontSize = 28.sp,
                fontWeight = FontWeight.Bold,
                color = Color.White,
                textAlign = TextAlign.Center
            )
            Spacer(modifier = Modifier.height(24.dp))

            Card(
                shape = RoundedCornerShape(20.dp),
                colors = CardDefaults.cardColors(containerColor = Color.White.copy(alpha = 0.94f)),
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(modifier = Modifier.padding(20.dp)) {
                    Text(text = "Untuk siapa ucapan ini?", fontWeight = FontWeight.Medium, fontSize = 14.sp)
                    Spacer(modifier = Modifier.height(8.dp))
                    OutlinedTextField(
                        value = name,
                        onValueChange = { name = it; showMessage = false },
                        placeholder = { Text("Masukkan nama") },
                        singleLine = true,
                        keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                        modifier = Modifier.fillMaxWidth()
                    )
                    Spacer(modifier = Modifier.height(16.dp))
                    Button(
                        onClick = { showMessage = true },
                        enabled = name.isNotBlank(),
                        modifier = Modifier.fillMaxWidth(),
                        colors = ButtonDefaults.buttonColors(containerColor = GradientTop)
                    ) {
                        Text("Kirim Ucapan \uD83C\uDF81")
                    }

                    AnimatedVisibility(visible = showMessage) {
                        Column(modifier = Modifier.padding(top = 16.dp)) {
                            Divider()
                            Spacer(modifier = Modifier.height(12.dp))
                            Text(text = "Selamat ulang tahun, $name! \uD83C\uDF89", fontWeight = FontWeight.Bold, fontSize = 18.sp)
                            Spacer(modifier = Modifier.height(4.dp))
                            Text(
                                text = "Semoga panjang umur, sehat selalu, dan semua harapan tahun ini terwujud. Selamat merayakan! \uD83E\uDD73",
                                fontSize = 14.sp,
                                color = Color.DarkGray
                            )
                        }
                    }
                }
            }
        }
    }
}

fun DiceWithButtonAndImage(modifier: Modifier = Modifier) {

    val imageResource = when(result) {

        3 -> R.drawable.dice_3
        4 -> R.drawable.dice_4
        5 -> R.drawable.dice_5

        Button(

        }
    }

                    .clip(RoundedCornerShape(16.dp))
            )

            Spacer(modifier = Modifier.height(14.dp))

            Text(
                text = "Nasi Goreng Spesial",
                fontSize = 20.sp,
                fontWeight = FontWeight.Bold,
                color = TextDark
            )
            Text(
                text = formatRupiah(price),
                fontSize = 16.sp,
                fontWeight = FontWeight.Bold,
                color = TextDark
            )
            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = "Nasi goreng dengan telur, ayam, dan sayuran segar.",
                fontSize = 14.sp,
                color = TextMuted
            )

            Spacer(modifier = Modifier.height(20.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                OutlinedButton(
                    onClick = { if (quantity > 1) quantity-- },
                    modifier = Modifier.size(48.dp),
                    shape = RoundedCornerShape(12.dp),
                    contentPadding = PaddingValues(0.dp)
                ) {
                    Text(text = "\u2212", fontSize = 22.sp)
                }

                Text(
                    text = quantity.toString(),
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier.width(72.dp),
                    color = TextDark,
                    textAlign = androidx.compose.ui.text.style.TextAlign.Center
                )

                Button(
                    onClick = { quantity++ },
                    modifier = Modifier.size(48.dp),
                    shape = RoundedCornerShape(12.dp),
                    colors = ButtonDefaults.buttonColors(containerColor = BluePrimary),
                    contentPadding = PaddingValues(0.dp)
                ) {
                    Text(text = "+", fontSize = 22.sp)
                }
            }

            Spacer(modifier = Modifier.height(16.dp))

            Button(
                onClick = {
                    cartCount += quantity
                    quantity = 1
                },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(50.dp),
                shape = RoundedCornerShape(12.dp),
                colors = ButtonDefaults.buttonColors(containerColor = GreenPrimary)
            ) {
                Icon(imageVector = Icons.Default.ShoppingCart, contentDescription = null)
                Spacer(modifier = Modifier.width(8.dp))
                Text(
                    text = "Tambah ke Keranjang \u2022 " + formatRupiah(price * quantity),
                    fontWeight = FontWeight.SemiBold
                )
            }

            Spacer(modifier = Modifier.height(10.dp))

            OutlinedButton(
                onClick = { showMenu = !showMenu },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(50.dp),
                shape = RoundedCornerShape(12.dp),
                border = BorderStroke(1.5.dp, BluePrimary)
            ) {
                Icon(
                    imageVector = Icons.Default.Menu,
                    contentDescription = null,
                    tint = BluePrimary
                )
                Spacer(modifier = Modifier.width(8.dp))
                Text(
                    text = if (showMenu) "Sembunyikan Menu" else "Lihat Menu",
                    color = BluePrimary,
                    fontWeight = FontWeight.SemiBold
                )
            }

            if (showMenu) {
                Spacer(modifier = Modifier.height(10.dp))
                MenuList()
            }

            Spacer(modifier = Modifier.height(10.dp))

            OutlinedButton(
                onClick = { isFavorite = !isFavorite },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(50.dp),
                shape = RoundedCornerShape(12.dp),
                border = BorderStroke(1.5.dp, RedPrimary),
                colors = ButtonDefaults.outlinedButtonColors(
                    containerColor = if (isFavorite) RedPrimary.copy(alpha = 0.10f) else Color.Transparent
                )
            ) {
                Icon(
                    imageVector = if (isFavorite) Icons.Default.Favorite else Icons.Default.FavoriteBorder,
                    contentDescription = null,
                    tint = RedPrimary
                )
                Spacer(modifier = Modifier.width(8.dp))
                Text(
                    text = "Favorit",
                    color = RedPrimary,
                    fontWeight = FontWeight.SemiBold
                )
            }
        }

        Column(
            modifier = Modifier
                .fillMaxWidth()
                .background(Color(0xFFF9FAFB))
                .padding(16.dp)
        ) {
            Button(
                onClick = {  },
                enabled = cartCount > 0,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(52.dp),
                shape = RoundedCornerShape(12.dp),
                colors = ButtonDefaults.buttonColors(containerColor = TextDark)
            ) {
                Text(
                    text = "Checkout ($cartCount item)",
                    fontSize = 16.sp,
                    fontWeight = FontWeight.Bold
                )
            }
        }
    }
}

@Composable
fun MenuList() {
    val items = listOf(
        "Nasi Goreng Spesial" to 25000,
        "Mie Ayam Bakso" to 22000,
        "Soto Ayam" to 20000,
        "Es Teh Manis" to 6000
    )

    Card(
        modifier = Modifier.fillMaxWidth(),
        shape = RoundedCornerShape(12.dp),
        colors = CardDefaults.cardColors(containerColor = Color(0xFFF3F4F6))
    ) {
        Column(modifier = Modifier.padding(14.dp)) {
            items.forEachIndexed { index, (name, price) ->
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 6.dp),
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    Text(text = name, color = TextDark, fontSize = 14.sp)
                    Text(
                        text = formatRupiah(price),
                        color = TextMuted,
                        fontSize = 14.sp
                    )
                }
                if (index < items.lastIndex) {
                    Spacer(modifier = Modifier.height(2.dp))
                }
            }
        }
    }
}
